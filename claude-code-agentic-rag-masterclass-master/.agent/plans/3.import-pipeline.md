### Task 3.2 — 建立 `chunking_service.py`

基於 token 數量的滑動視窗分塊器：

```python
# backend/app/services/chunking_service.py
from dataclasses import dataclass
import tiktoken

CHUNK_SIZE = 512       # tokens per chunk
CHUNK_OVERLAP = 64     # overlap between consecutive chunks
ENCODING = "cl100k_base"  # used by OpenAI models; compatible with most providers


@dataclass
class Chunk:
    content: str
    chunk_index: int
    token_count: int


def chunk_text(text: str, chunk_size: int = CHUNK_SIZE, overlap: int = CHUNK_OVERLAP) -> list[Chunk]:
    """
    Split text into overlapping token-based chunks.
    Returns list of Chunk objects with index and token count.
    """
    enc = tiktoken.get_encoding(ENCODING)
    tokens = enc.encode(text)

    chunks = []
    start = 0
    chunk_index = 0

    while start < len(tokens):
        end = min(start + chunk_size, len(tokens))
        chunk_tokens = tokens[start:end]
        content = enc.decode(chunk_tokens)

        chunks.append(Chunk(
            content=content,
            chunk_index=chunk_index,
            token_count=len(chunk_tokens),
        ))

        chunk_index += 1
        start += chunk_size - overlap  # slide with overlap

        if end == len(tokens):
            break

    return chunks
```

### Task 3.3 — 建立 `embedding_service.py`

```python
# backend/app/services/embedding_service.py
from app.services.llm_service import get_embedding_client
from app.config import settings

BATCH_SIZE = 512  # OpenAI max is 2048; use 512 for safety


def embed_texts(texts: list[str]) -> list[list[float]]:
    """
    Embed a list of texts using the configured embedding model.
    Handles batching automatically.
    Returns list of embedding vectors in the same order as input.
    """
    client = get_embedding_client()
    all_embeddings = []

    for i in range(0, len(texts), BATCH_SIZE):
        batch = texts[i:i + BATCH_SIZE]
        response = client.embeddings.create(
            model=settings.EMBEDDING_MODEL,
            input=batch,
        )
        # Sort by index to ensure order is preserved
        sorted_data = sorted(response.data, key=lambda x: x.index)
        all_embeddings.extend([item.embedding for item in sorted_data])

    return all_embeddings
```

### Task 3.4 — 建立 `import_service.py`

完整管線：提取 → 分塊 → 嵌入 → 儲存，以 BackgroundTask 執行：

```python
# backend/app/services/import_service.py
from supabase import Client
from app.services.chunking_service import chunk_text
from app.services.embedding_service import embed_texts


def _extract_text(content: bytes, file_type: str) -> str:
    """
    Extract plain text from file bytes.
    Module 2: supports txt and md only.
    Module 5 will add pdf, docx, html via docling.
    """
    if file_type in ("txt", "md"):
        return content.decode("utf-8", errors="replace")
    raise ValueError(f"Unsupported file type for Module 2: {file_type}")


def process_document(
    document_id: str,
    user_id: str,
    file_content: bytes,
    file_type: str,
    supabase: Client,
) -> None:
    """
    Full pipeline: extract → chunk → embed → store.
    Called as a background task. Updates document.status at each stage.
    Supabase Realtime broadcasts status changes to subscribed frontend clients.
    """
    def _set_status(status: str, error_msg: str | None = None):
        update = {"status": status}
        if error_msg:
            update["error_msg"] = error_msg
        supabase.table("documents").update(update).eq("id", document_id).execute()

    try:
        _set_status("processing")

        # Stage 1: Extract text
        text = _extract_text(file_content, file_type)

        # Stage 2: Chunk
        chunks = chunk_text(text)

        # Stage 3: Embed all chunks in one batch call
        texts = [c.content for c in chunks]
        embeddings = embed_texts(texts)

        # Stage 4: Batch insert chunks
        rows = [
            {
                "document_id": document_id,
                "user_id": user_id,
                "content": chunk.content,
                "chunk_index": chunk.chunk_index,
                "token_count": chunk.token_count,
                "embedding": embedding,
            }
            for chunk, embedding in zip(chunks, embeddings)
        ]

        BATCH = 100
        for i in range(0, len(rows), BATCH):
            supabase.table("chunks").insert(rows[i:i + BATCH]).execute()

        # Stage 5: Mark complete
        supabase.table("documents").update({
            "status": "complete",
            "chunk_count": len(chunks),
        }).eq("id", document_id).execute()

    except Exception as e:
        _set_status("error", error_msg=str(e))
        raise
```

**驗證：** 上傳小型 `.txt` 檔後，documents 表的 status 應從 `pending → processing → complete`，chunk_count 應大於 0。

### Task 3.5 — 建立 `routers/documents.py`

```python
# backend/app/routers/documents.py
from fastapi import APIRouter, Depends, HTTPException, UploadFile, File, BackgroundTasks
from app.dependencies import get_current_user, get_supabase_client
from app.services.import_service import process_document

router = APIRouter()

ALLOWED_TYPES = {"txt", "md"}       # Module 5 擴充 pdf/docx/html
MAX_FILE_SIZE = 10 * 1024 * 1024    # 10 MB


@router.get("/documents")
async def list_documents(user: dict = Depends(get_current_user)):
    supabase = get_supabase_client()
    result = supabase.table("documents").select("*") \
        .eq("user_id", user["id"]).order("created_at", desc=True).execute()
    return result.data


@router.post("/documents")
async def upload_document(
    background_tasks: BackgroundTasks,
    file: UploadFile = File(...),
    user: dict = Depends(get_current_user),
):
    ext = (file.filename or "").rsplit(".", 1)[-1].lower()
    if ext not in ALLOWED_TYPES:
        raise HTTPException(
            status_code=400,
            detail=f"Unsupported file type: .{ext}. Allowed: {sorted(ALLOWED_TYPES)}"
        )

    content = await file.read()
    if len(content) > MAX_FILE_SIZE:
        raise HTTPException(status_code=413, detail="File too large. Max 10 MB.")

    supabase = get_supabase_client()

    result = supabase.table("documents").insert({
        "user_id": user["id"],
        "filename": file.filename,
        "file_type": ext,
        "file_size": len(content),
        "status": "pending",
    }).execute()
    document = result.data[0]

    background_tasks.add_task(
        process_document,
        document_id=document["id"],
        user_id=user["id"],
        file_content=content,
        file_type=ext,
        supabase=supabase,
    )

    return document


@router.delete("/documents/{document_id}")
async def delete_document(document_id: str, user: dict = Depends(get_current_user)):
    supabase = get_supabase_client()
    supabase.table("documents").delete() \
        .eq("id", document_id).eq("user_id", user["id"]).execute()
    return {"message": "Deleted"}
```

### Task 3.8 — 新增 `uploadFile` helper 到 `api.ts`

**重要：** FormData 上傳時，瀏覽器需自動設置 `Content-Type: multipart/form-data; boundary=...`，不可手動設置。

```typescript
// frontend/src/lib/api.ts — 新增
import { supabase } from '@/lib/supabase'

export async function uploadFile(path: string, formData: FormData): Promise<Response> {
  const { data: { session } } = await supabase.auth.getSession()
  const headers: Record<string, string> = {}
  if (session?.access_token) {
    headers['Authorization'] = `Bearer ${session.access_token}`
  }
  // 注意：不設置 Content-Type，讓瀏覽器自動加上 multipart boundary
  const res = await fetch(`/api${path}`, { method: 'POST', headers, body: formData })
  if (!res.ok) {
    const error = await res.json().catch(() => ({ detail: 'Upload failed' }))
    throw new Error(error.detail || 'Upload failed')
  }
  return res
}
```

### Task 3.11 — 建立 `ImportPage` + 導航

```typescript
// frontend/src/pages/ImportPage.tsx
import { useState, useEffect, useCallback } from 'react'
import { supabase } from '@/lib/supabase'
import { uploadFile, apiFetch } from '@/lib/api'
import DropZone from '@/components/import/DropZone'
import DocumentList from '@/components/import/DocumentList'

interface ImportPageProps {
  onNavigateToChat: () => void
}

export default function ImportPage({ onNavigateToChat }: ImportPageProps) {
  const [documents, setDocuments] = useState<Document[]>([])
  const [uploading, setUploading] = useState(false)

  // 載入現有 documents
  const fetchDocuments = useCallback(async () => {
    const res = await apiFetch('/documents')
    setDocuments(await res.json())
  }, [])

  useEffect(() => { fetchDocuments() }, [fetchDocuments])

  // 訂閱 Realtime 更新
  useEffect(() => {
    const channel = supabase
      .channel('documents-changes')
      .on('postgres_changes', { event: 'UPDATE', schema: 'public', table: 'documents' },
        (payload) => {
          setDocuments(prev =>
            prev.map(doc => doc.id === payload.new.id ? { ...doc, ...payload.new } : doc)
          )
        }
      )
      .subscribe()
    return () => { supabase.removeChannel(channel) }
  }, [])

  const handleUpload = async (files: File[]) => {
    setUploading(true)
    for (const file of files) {
      const formData = new FormData()
      formData.append('file', file)
      try {
        const res = await uploadFile('/documents', formData)
        const doc = await res.json()
        setDocuments(prev => [doc, ...prev])
      } catch (e) {
        console.error('Upload failed:', e)
      }
    }
    setUploading(false)
  }

  const handleDelete = async (id: string) => {
    await apiFetch(`/documents/${id}`, { method: 'DELETE' })
    setDocuments(prev => prev.filter(d => d.id !== id))
  }

  // ...render DropZone + DocumentList
}
```

## Validation Tests

1. 上傳 `.txt` 文件 → document 卡片立即顯示 `pending`，透過 Realtime 切換到 `processing` 再到 `complete`
2. 上傳 `.pdf` 文件 → 後端回傳 HTTP 400
3. 上傳超過 10 MB 的文件 → HTTP 413
4. 上傳成功後，在 Supabase SQL Editor 驗證：
   ```sql
   SELECT count(*) FROM chunks WHERE document_id = '<id>';
   -- 應等於 document.chunk_count
   ```
5. 驗證 embedding 已儲存：
   ```sql
   SELECT embedding IS NOT NULL FROM chunks LIMIT 5;
   -- 所有 row 應為 true
   ```
6. 刪除 document → chunks 自動刪除（CASCADE）：
   ```sql
   SELECT count(*) FROM chunks WHERE document_id = '<deleted_id>';
   -- 應為 0
   ```
