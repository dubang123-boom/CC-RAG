# Module 3: Record Manager (Content Hashing + Deduplication)

## Complexity Rating

✅ **Simple** - 单次可执行，低风险。所有变更都是追加式的（一个新 SQL 列、一个新索引、上传端点中的分支逻辑）。无需创建新文件。

---

## Architecture Overview

去重决策完全在 `upload_document`（POST 端点）中、插入 `documents` 表**之前**完成。后台任务 `process_document` 无需修改。

```
Upload request
    → read file bytes
    → compute SHA-256 hash
    → query documents WHERE user_id = ? AND content_hash = ?
        Case A: found, status = "complete"       → HTTP 409 (duplicate)
        Case B: found, status = "error"          → reset to pending, re-process
        Case C: found, status = pending/processing → HTTP 409 (already processing)
        Case D: NOT found by hash, filename matches (changed) → delete old doc+chunks, insert new
        Case E: not found at all                 → insert new (normal path)
    → add BackgroundTask to process_document
    → return document row
```

### Step 3: `backend/app/routers/documents.py` (full rewrite)

```python
import hashlib
from fastapi import APIRouter, Depends, HTTPException, UploadFile, File, BackgroundTasks
from app.dependencies import get_current_user, get_supabase_client
from app.services.import_service import process_document

router = APIRouter()

ALLOWED_TYPES = {"txt", "md"}
MAX_FILE_SIZE = 10 * 1024 * 1024  # 10 MB


def _compute_hash(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


@router.get("/documents")
async def list_documents(user: dict = Depends(get_current_user)):
    supabase = get_supabase_client()
    result = supabase.table("documents").select("*") \
        .eq("user_id", user["id"]).order("created_at", desc=True).execute()
    return result.data


@router.post("/documents")
async def upload_document(
    background_tasks: BackgroundTasks,
    file: UploadFile = File(...),
    user: dict = Depends(get_current_user),
):
    ext = (file.filename or "").rsplit(".", 1)[-1].lower()
    if ext not in ALLOWED_TYPES:
        raise HTTPException(
            status_code=400,
            detail=f"Unsupported file type: .{ext}. Allowed: {sorted(ALLOWED_TYPES)}",
        )

    content = await file.read()
    if len(content) > MAX_FILE_SIZE:
        raise HTTPException(status_code=413, detail="File too large. Max 10 MB.")

    content_hash = _compute_hash(content)
    user_id = user["id"]
    supabase = get_supabase_client()

    # 1. Check for exact hash match
    hash_match = (
        supabase.table("documents")
        .select("id, status, filename")
        .eq("user_id", user_id)
        .eq("content_hash", content_hash)
        .execute()
    )

    if hash_match.data:
        existing = hash_match.data[0]

        if existing["status"] == "complete":
            raise HTTPException(
                status_code=409,
                detail=(
                    f"Duplicate: this file's content already exists as "
                    f'"{existing["filename"]}" (id: {existing["id"]}).'
                ),
            )

        if existing["status"] == "error":
            supabase.table("documents").update({
                "status": "pending",
                "error_msg": None,
                "filename": file.filename,
                "file_size": len(content),
            }).eq("id", existing["id"]).execute()

            document = (
                supabase.table("documents")
                .select("*")
                .eq("id", existing["id"])
                .single()
                .execute()
            ).data

            background_tasks.add_task(
                process_document,
                document_id=document["id"],
                user_id=user_id,
                file_content=content,
                file_type=ext,
                supabase=supabase,
            )
            return document

        raise HTTPException(
            status_code=409,
            detail=(
                f"File is already being processed "
                f'(id: {existing["id"]}, status: {existing["status"]}).'
            ),
        )

    # 2. Same filename, different content → update (delete old)
    name_match = (
        supabase.table("documents")
        .select("id")
        .eq("user_id", user_id)
        .eq("filename", file.filename)
        .execute()
    )

    if name_match.data:
        old_id = name_match.data[0]["id"]
        supabase.table("documents").delete().eq("id", old_id).execute()

    # 3. Insert new document row
    result = supabase.table("documents").insert({
        "user_id": user_id,
        "filename": file.filename,
        "file_type": ext,
        "file_size": len(content),
        "status": "pending",
        "content_hash": content_hash,
    }).execute()
    document = result.data[0]

    background_tasks.add_task(
        process_document,
        document_id=document["id"],
        user_id=user_id,
        file_content=content,
        file_type=ext,
        supabase=supabase,
    )

    return document


@router.delete("/documents/{document_id}")
async def delete_document(document_id: str, user: dict = Depends(get_current_user)):
    supabase = get_supabase_client()
    supabase.table("documents").delete() \
        .eq("id", document_id).eq("user_id", user["id"]).execute()
    return {"message": "Deleted"}
```

## Validation Tests

1. **Migration applied**: 在 Supabase SQL Editor 中确认 `content_hash` 列存在
2. **正常上传**: 上传文件后，DB 中 `content_hash` 为 64 位十六进制字符串
3. **重复拒绝**: 再次上传同一文件 → HTTP 409 含 "Duplicate"
4. **文件更新**: 上传同名但内容不同的文件 → HTTP 200，旧 document+chunks 已删除
5. **用户隔离**: 不同用户上传同一文件 → 各自独立，不触发 409
